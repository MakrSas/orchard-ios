# Meson cross-file: arm64 iOS (device), for cross-compiling qemu/ as a
# shared library (see meson_options.txt's `shared_lib`) that ios/app dlopens.
#
# Modelled on the cross-file Inferno-iOS uses for its own QEMU fork — same
# mechanism, same reused build of glib/gmp/pixman/lzfse/nettle/libpng/
# libslirp/libtasn1/libucontext (generic C deps, not iPhone-specific), at
# ~/inferno-ios/prefix. Regenerate the SDK path with
# `xcrun --sdk iphoneos --show-sdk-path` if it stops matching your Xcode.

ranlib     = '/Applications/Xcode.app/Contents/Developer/Toolchains/XcodeDefault.xctoolchain/usr/bin/ranlib'

# Which configs/devices/<target>/<name>.mak meson.build's own minikconf step
# reads (meson.get_external_property(target, 'default')) — the equivalent of
# configure's --with-devices-aarch64=ios, which only exists as a flag to
# configure itself, not as something a hand-written cross-file inherits for
# free. Selects ios.mak (REIMS_VGPU off; see that file for why) over
# default.mak.
aarch64-softmmu = 'ios'
